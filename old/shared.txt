// ==UserScript==
// @name         MNQ Shared
// @namespace    http://tampermonkey.net/
// @version      1.0
// @description  Shared methods
// @author       You
// @match        https://www.tradingview.com/chart/*
// @icon         https://s3.tradingview.com/userpics/6171439-mFQX_big.png
// @grant        GM_xmlhttpRequest
// @grant        unsafeWindow
// @require      https://cdnjs.cloudflare.com/ajax/libs/moment.js/2.30.1/moment.min.js
// @require      https://cdnjs.cloudflare.com/ajax/libs/moment-timezone/0.5.45/moment-timezone-with-data.min.js
// @connect      script.google.com
// @connect      script.googleusercontent.com
// @connect      *
// @run-at       document-start
// ==/UserScript==

(function() {
    'use strict';
    const rootScope = unsafeWindow;

    const TELEGRAM_GAS_PROXY = 'https://script.google.com/macros/s/AKfycbz3X6EE6OJIWMCX8TsxyTKGtjABMihv8ygf0V7EYTpAaEWnp3ffSQNf3Ouxfix7bSMf/exec';
    const TELEGRAM_BOT_TOKEN = '8721637113:AAE6LY0BgBIYtcsqdo29I2nyp0e63PnTmzo';
    const TELEGRAM_CHAT_ID = '7670215141';

    const LOGGER_GAS_PROXY = 'https://script.google.com/macros/s/AKfycbz3X6EE6OJIWMCX8TsxyTKGtjABMihv8ygf0V7EYTpAaEWnp3ffSQNf3Ouxfix7bSMf/exec';
    const LOGGER_BOT_TOKEN = '8916407832:AAHjG7jtmP4_gdhPgsaika8P0yGMmKZb-I4';
    const LOGGER_CHAT_ID = '7670215141';

    // State Variables
    rootScope.isRequestPending = false;
    rootScope.isLoggerPending = false;

    /**
     * Telegram Webhook notification handler
     */
    rootScope.sendTelegramAlert = function(customMessage) {
        if (rootScope.isRequestPending) return;
        rootScope.isRequestPending = true;

        const tz = moment().tz("Asia/Karachi");
        const message = customMessage || `Setup triggered at ${tz.format('hh:mm:ss A')}`;
        console.log(customMessage ? customMessage : 'Sending..');

        return new Promise((resolve) => {
            GM_xmlhttpRequest({
                method: "POST",
                url: TELEGRAM_GAS_PROXY,
                headers: {
                    "Content-Type": "text/plain"
                },
                data: JSON.stringify({
                    token: TELEGRAM_BOT_TOKEN,
                    chat_id: TELEGRAM_CHAT_ID,
                    text: message
                }),
                onload: () => {
                    resolve();
                    rootScope.isRequestPending = false;
                },
                onerror: () => {
                    resolve();
                    rootScope.isRequestPending = false;
                },
                ontimeout: () => {
                    resolve();
                    rootScope.isRequestPending = false;
                },
            });
        });
    }

    /**
     * Telegram Webhook notification handler
     */
    rootScope.sendLogs = function(from) {
        if (rootScope.isLoggerPending) return;
        rootScope.isLoggerPending = true;

        const tz = moment().tz("Asia/Karachi");
        const pref = from ? from : 'Ping';
        const message = `${pref}: ${tz.format('hh:mm:ss A')}`;
        // console.log('Logging..');

        return new Promise((resolve) => {
            GM_xmlhttpRequest({
                method: "POST",
                url: LOGGER_GAS_PROXY,
                headers: {
                    "Content-Type": "text/plain"
                },
                data: JSON.stringify({
                    token: LOGGER_BOT_TOKEN,
                    chat_id: LOGGER_CHAT_ID,
                    text: message
                }),
                onload: () => {
                    resolve();
                    rootScope.isLoggerPending = false;
                },
                onerror: () => {
                    resolve();
                    rootScope.isLoggerPending = false;
                },
                ontimeout: () => {
                    resolve();
                    rootScope.isLoggerPending = false;
                },
            });
        });
    }

    // Initial console
    console.log('✔ Methods loaded!');
})();
