function authorizeUrlFetch() {
  UrlFetchApp.fetch("https://api.telegram.org");
}

function sendLogs(data) {
  try {
    var token = data.token;
    var userId = data.chat_id;
    var text = data.text;
    
    var telegramUrl = "https://api.telegram.org/bot" + token + "/sendMessage";
    
    var payload = {
      "chat_id": userId,
      "text": text
    };
    
    var options = {
      "method": "post",
      "contentType": "application/json",
      "payload": JSON.stringify(payload),
      "muteHttpExceptions": true
    };
    
    var response = UrlFetchApp.fetch(telegramUrl, options);
    return ContentService.createTextOutput(response.getContentText()).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({"ok": false, "error": err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}

// POST Requests ke liye
function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  return sendLogs(data);
}

  var data = {
