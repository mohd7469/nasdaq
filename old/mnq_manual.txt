// ==UserScript==
// @name         MNQ Manual Execution
// @namespace    http://tampermonkey.net/
// @version      1.0
// @description  For manual trade execution
// @author       You
// @match        https://www.tradingview.com/chart/*
// @icon         https://s3.tradingview.com/userpics/6171439-mFQX_big.png
// @grant        GM_xmlhttpRequest
// @grant        unsafeWindow
// @require      https://cdnjs.cloudflare.com/ajax/libs/moment.js/2.30.1/moment.min.js
// @require      https://cdnjs.cloudflare.com/ajax/libs/moment-timezone/0.5.45/moment-timezone-with-data.min.js
// @connect      script.google.com
// @connect      script.googleusercontent.com
// @connect      *
// @run-at       document-idle
// ==/UserScript==

(async function() {
    'use strict';
    const rootScope = unsafeWindow;

    // =========================================================================
    // 1. TOP-LEVEL CONFIGURATION / SEPARATE VARIABLES
    // =========================================================================
    const CHECK_INTERVAL_MS = 4000;
    const alertAudio = new Audio('https://cdn.jsdelivr.net/gh/mohd7469/alert@main/beep1.mp3');

    // Notification Permission Check / Request
    if (typeof Notification !== 'undefined' && Notification.permission !== 'granted') {
        Notification.requestPermission();
    }

    // Separate Sessions Configuration
    const OVERNIGHT = {
        startTime: '03:00',
        endTime: '12:00',
        change: 50,
        volume: 5000
    };

    const LONDON = {
        startTime: '12:00',
        endTime: '18:30',
        change: 50,
        volume: 10000
    };

    const NEW_YORK = {
        startTime: '18:30',
        endTime: '01:00',
        change: 50,
        volume: 50000
    };


    // =========================================================================
    // 2. HELPER METHODS
    // =========================================================================

    /**
     * Connect Tradingview Session
     */
    function connectTradingviewSession() {
        let el = document.querySelector('[data-overflow-tooltip-text="Connect"]');
        if (el !== null && typeof el.click === 'function') {
            el.click();
            console.log('Reconnecting..');
        }
    }

    /**
     * Chrome Desktop Push Notification Function
     */
    function showDesktopAlert() {
        if (typeof Notification !== 'undefined' && Notification.permission === 'granted') {
            const tz = moment().tz("Asia/Karachi");

            const notif = new Notification('Nasdaq Alert', {
                body: `Setup triggered at ${tz.format('hh:mm:ss A')}`,
                icon: 'https://www.nasdaq.com/sites/acquia.prod/files/2020/09/24/nasdaq.jpg'
            });

            notif.onclick = function() {
                window.focus();
                notif.close();
            };
        }
    }

    /**
     * DOM se title ke basis par clean float value nikalne ke liye
     */
    function getValueByTitle(title) {
        const parent = document.querySelector(`[data-test-id-value-title="${title}"]`);
        if (!parent) return null;

        const valueElement = parent.querySelector('[class*="valueValue-"]') || parent;
        let text = valueElement.innerText.replace(/,/g, '').trim();

        let multiplier = 1;
        if (text.endsWith('K') || text.endsWith('k')) multiplier = 1000;
        else if (text.endsWith('M') || text.endsWith('m')) multiplier = 1000000;

        if (multiplier > 1) {
            text = text.slice(0, -1);
        }

        const value = parseFloat(text);
        return isNaN(value) ? null : value * multiplier;
    }

    /**
     * Number ko formatted K, M, B representation me convert karne ke liye
     */
    function formatVolume(input) {
        if (input === null || input === undefined) return '0';

        let num = input;
        if (typeof num === 'string') {
            let text = num.replace(/,/g, '').trim();
            let multiplier = 1;

            if (text.endsWith('K') || text.endsWith('k')) multiplier = 1000;
            else if (text.endsWith('M') || text.endsWith('m')) multiplier = 1000000;
            else if (text.endsWith('B') || text.endsWith('b')) multiplier = 1000000000;

            if (multiplier > 1) text = text.slice(0, -1);
            num = parseFloat(text) * multiplier;
        }

        if (isNaN(num)) return '0';

        const absNum = Math.abs(num);
        if (absNum < 1000) return num.toString();
        if (absNum < 1000000) return (num / 1000).toFixed(2).replace(/\.00$/, '').replace(/(\.\d)0$/, '$1') + 'k';
        if (absNum < 1000000000) return (num / 1000000).toFixed(2).replace(/\.00$/, '').replace(/(\.\d)0$/, '$1') + 'M';
        return (num / 1000000000).toFixed(2).replace(/\.00$/, '').replace(/(\.\d)0$/, '$1') + 'B';
    }

    /**
     * "HH:mm" string ko day minutes (0-1439) me convert karta hai
     */
    function parseTimeToMinutes(timeStr) {
        const [hours, minutes] = timeStr.split(':').map(Number);
        return hours * 60 + minutes;
    }

    /**
     * Native Date API ka use karke exact session dynamic thresholds nikalna
     */
    function getDynamicTargets() {
        const tz = moment().tz("Asia/Karachi");
        const currentMinutes = tz.hours() * 60 + tz.minutes();

        const overnightStart = parseTimeToMinutes(OVERNIGHT.startTime);
        const overnightEnd   = parseTimeToMinutes(OVERNIGHT.endTime);

        const londonStart    = parseTimeToMinutes(LONDON.startTime);
        const londonEnd      = parseTimeToMinutes(LONDON.endTime);

        const nyStart        = parseTimeToMinutes(NEW_YORK.startTime);
        const nyEnd          = parseTimeToMinutes(NEW_YORK.endTime);

        // Overnight check (03:00 - 12:00)
        if (currentMinutes >= overnightStart && currentMinutes < overnightEnd) {
            return { targetChange: OVERNIGHT.change, targetVolume: OVERNIGHT.volume };
        }
        // London check (12:00 - 18:30)
        else if (currentMinutes >= londonStart && currentMinutes < londonEnd) {
            return { targetChange: LONDON.change, targetVolume: LONDON.volume };
        }
        // New York check (18:30 - 01:00 Next Day)
        else if (currentMinutes >= nyStart || currentMinutes < nyEnd) {
            return { targetChange: NEW_YORK.change, targetVolume: NEW_YORK.volume };
        }

        return { targetChange: 50, targetVolume: 5000 };
    }


    // =========================================================================
    // 3. BACKGROUND WORKER & MAIN LOOP EXECUTION
    // =========================================================================

    const workerCode = `
        let intervalId = null;
        self.onmessage = function(e) {
            if (e.data.action === 'start') {
                intervalId = setInterval(() => {
                    self.postMessage('tick');
                }, e.data.interval);
            } else if (e.data.action === 'stop') {
                clearInterval(intervalId);
            }
        };
    `;

    const blob = new Blob([workerCode], { type: 'application/javascript' });
    const worker = new Worker(URL.createObjectURL(blob));

    // Flag for preventing multiple reload triggers within the same minute
    let isReloading = false;

    worker.onmessage = async function(e) {
        await rootScope.sendLogs('Ping(M)');
        connectTradingviewSession();

        if (e.data !== 'tick') return;

        // ---------------------------------------------------------------------
        // MIDNIGHT RELOAD CHECK (00:00:00 - 00:00:59)
        // ---------------------------------------------------------------------
        const tz = moment().tz("Asia/Karachi");
        if (tz.hours() === 0 && tz.minutes() === 0 && !isReloading) {
            isReloading = true;
            console.log('[Midnight] Reloading server...');
            await rootScope.sendLogs('Midnight(M)');
            setTimeout(() => {
                location.reload();
            }, 10000);
            return;
        }

        const open = getValueByTitle('O');
        const close = getValueByTitle('C');
        const volume = getValueByTitle('Vol');

        if (open === null || close === null || volume === null) {
            console.log('Waiting for data..');
            return;
        }

        const diff = Math.abs(close - open);
        const formattedDiffNum = parseFloat(diff.toFixed(2));

        const { targetChange, targetVolume } = getDynamicTargets();

        // Screen Output Formatter
        const sessionChange = `$${formatVolume(targetChange)}`.padEnd(7, ' ');
        const sessionVol = formatVolume(targetVolume).padEnd(7, ' ');
        const currentChange = `${diff.toFixed(2)}`.padEnd(7, ' ');
        const currentVol = formatVolume(volume).padEnd(7, ' ');

        console.log(`change: ${sessionChange} |  ${currentChange}\nvolume: ${sessionVol} |  ${currentVol}`);

         // Trigger Check
        if (formattedDiffNum >= targetChange && volume >= targetVolume) {
            try {
                await alertAudio.play();
                console.log('Ringing..');
            } catch (err) {
                console.warn('Audio play error', err);
            }
            showDesktopAlert();
            await rootScope.sendTelegramAlert();
        }

        rootScope.isRequestPending = false
    };

    // Execution Trigger
    worker.postMessage({ action: 'start', interval: CHECK_INTERVAL_MS });

    // Global Expose to Stop Worker manually
    window.stopWorker = function() {
        worker.postMessage({ action: 'stop' });
        worker.terminate();
        console.log('✔ Worker Stopped!');
    };

    await rootScope.sendTelegramAlert('Manual Execution Started!');
    await rootScope.sendLogs('Ping(M)');
})();
