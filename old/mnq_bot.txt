// ==UserScript==
// @name         MNQ Bot Execution
// @namespace    http://tampermonkey.net/
// @version      1.0
// @description  For automatic bot execution
// @author       You
// @match        https://www.tradingview.com/chart/*
// @icon         https://s3.tradingview.com/userpics/6171439-mFQX_big.png
// @grant        GM_xmlhttpRequest
// @grant        unsafeWindow
// @require      https://cdnjs.cloudflare.com/ajax/libs/moment.js/2.30.1/moment.min.js
// @require      https://cdnjs.cloudflare.com/ajax/libs/moment-timezone/0.5.45/moment-timezone-with-data.min.js
// @connect      script.google.com
// @connect      script.googleusercontent.com
// @connect      *
// @run-at       document-idle
// ==/UserScript==

(async function() {
    'use strict';
    const rootScope = unsafeWindow;

    // =========================================================================
    // 1. TOP-LEVEL CONFIGURATION / SEPARATE VARIABLES
    // =========================================================================
    const CHECK_INTERVAL_MS = 1000;

    // DOM Selectors
    const CONTAINER_SELECTOR = '[data-test-id-widget-type="pine_logs"] [class*="logsList-"]';
    const SCROLL_BUTTON_SELECTOR = 'button';
    const LOG_ITEM_SELECTOR = 'div[data-index]';

    // State variable to avoid duplicates
    let lastProcessedLogText = "";

    // =========================================================================
    // 2. HELPER METHODS
    // =========================================================================

    /**
     * Target container ke andar scroll button dhoond kar click karta hai
     * @param {Element} container
     */
    function clickScrollButton(container) {
        if (!container) return;
        const button = container.querySelector(SCROLL_BUTTON_SELECTOR);
        if (button) {
            button.click();
        }
    }

    /**
     * Container se sabse aakhri data-index element (latest log) ko pick karta hai
     * @param {Element} container
     * @returns {Element|null}
     */
    function getLatestLogElement(container) {
        if (!container) return null;
        const logElements = container.querySelectorAll(LOG_ITEM_SELECTOR);
        if (logElements.length === 0) return null;
        return logElements[logElements.length - 1];
    }

    /**
     * Scraped raw text ko parse karke clean JSON object banata hai.
     * UserScript me included Moment.js library ka use karke date formatting karta hai.
     *
     * @param {string} rawData
     * @returns {Object|null}
     */
    function parseLogData(rawData) {
        if (!rawData || !rawData.trim()) return null;

        // Date extraction [ISO_TIMESTAMP]
        const dateMatch = rawData.match(/\[(.*?)\]/);
        const isoDateString = dateMatch ? dateMatch[1] : null;

        // Action Matching: LONG_ENTRY, SHORT_ENTRY, LONG_EXIT, SHORT_EXIT
        const actionMatch = rawData.match(/(LONG|SHORT)_(ENTRY|EXIT):\s*([\d\.]+)/);
        if (!actionMatch) return null;

        const side = actionMatch[1];                                           // "LONG" | "SHORT"
        const action = actionMatch[2];                                         // "ENTRY" | "EXIT"
        const triggerPrice = parseFloat(actionMatch[3]);                       // Entry or Exit Price

        // Optional TP & SL values
        const tpMatch = rawData.match(/TP:\s*([\d\.]+)/);
        const slMatch = rawData.match(/SL:\s*([\d\.]+)/);

        const tpValue = tpMatch ? parseFloat(tpMatch[1]) : null;
        const slValue = slMatch ? parseFloat(slMatch[1]) : null;

        // Dynamic Hash generation for 'id'
        const generatedHash = `${side}_${action}_${triggerPrice}_${isoDateString}`;

        return {
            id: generatedHash,
            timestamp: isoDateString,
            side: side,                                                        // "LONG" | "SHORT"
            action: action,                                                    // "ENTRY" | "EXIT"
            price: triggerPrice,
            tp: tpValue,
            sl: slValue
        };
    }

    /**
     * Core handler
     */
    let processedHistory = new Set();

    function processPineLogsTask() {
        const logsContainer = document.querySelector(CONTAINER_SELECTOR);
        if (!logsContainer) {
            console.log('Waiting for data..');
            return;
        }

        // Task 1: Sub div[data-index] elements pick karain
        const allLogElements = logsContainer.querySelectorAll(LOG_ITEM_SELECTOR);
        if (!allLogElements || allLogElements.length === 0) return;

        // Task 2: Scroll button click karna
        clickScrollButton(logsContainer);

        // Aakhri 3-4 elements ka reference lein taake instant multi-log skip na ho
        const recentLogs = Array.from(allLogElements).slice(-3);

        recentLogs.forEach((logEl) => {
            const rawData = (logEl.innerText || logEl.textContent || "").trim();
            if (!rawData) return;

            // Parse clean object using rawData
            const jsonOutput = parseLogData(rawData);

            if (jsonOutput) {
                // Duplicate Hash Check
                if (!processedHistory.has(jsonOutput.id)) {
                    processedHistory.add(jsonOutput.id);

                    // Memory Cleanup (Max 50 Hashes)
                    if (processedHistory.size > 50) {
                        const oldestHash = processedHistory.values().next().value;
                        processedHistory.delete(oldestHash);
                    }

                    // Clean Verified Output
                    console.log(JSON.stringify(jsonOutput, null, 2));

                    // PHASE 2: Send to server
                    // sendToNodeServer(jsonOutput);
                }
            }
        });
    }

    // =========================================================================
    // 3. BACKGROUND WORKER & MAIN LOOP EXECUTION
    // =========================================================================

    const workerCode = `
        let intervalId = null;
        self.onmessage = function(e) {
            if (e.data.action === 'start') {
                intervalId = setInterval(() => {
                    self.postMessage('tick');
                }, e.data.interval);
            } else if (e.data.action === 'stop') {
                clearInterval(intervalId);
            }
        };
    `;

    const blob = new Blob([workerCode], { type: 'application/javascript' });
    const worker = new Worker(URL.createObjectURL(blob));
    // Flag for preventing multiple reload triggers within the same minute
    let isReloading = false;

    worker.onmessage = async function(e) {
        await rootScope.sendLogs('Ping(B)');

        if (e.data !== 'tick') return;

        // ---------------------------------------------------------------------
        // MIDNIGHT RELOAD CHECK (00:00:00 - 00:00:59)
        // ---------------------------------------------------------------------
        const tz = moment().tz("Asia/Karachi");
        if (tz.hours() === 0 && tz.minutes() === 0 && !isReloading) {
            isReloading = true;
            console.log('[Midnight] Reloading server...');
            await rootScope.sendLogs('Midnight(B)');
            setTimeout(() => {
                location.reload();
            }, 10000);
            return;
        }

        // Execute parsing loop on worker tick
        processPineLogsTask();
    };

    // Execution Trigger
    worker.postMessage({ action: 'start', interval: CHECK_INTERVAL_MS });

    // Global Expose to Stop Worker manually
    window.stopWorker = function() {
        worker.postMessage({ action: 'stop' });
        worker.terminate();
        console.log('✔ Worker Stopped!');
    };


    await rootScope.sendTelegramAlert('Bot Execution Started!');
    await rootScope.sendLogs('Ping(B)');
})();
